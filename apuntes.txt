Compliar: make

1. Probar crear un directorio .png
./mkfs.bwfs imagenes/

2. Probar mount 
mkdir punto
./mount.bwfs imagenes punto -f -d

otra consola: ls punto

3. Probar inject imagenes 
./mkfs.bwfs fs_data/
gcc -o tools/inject tools/inject.c -Iinclude -lpng
./tools/inject
mkdir punto_montaje
./mount.bwfs fs_data/ punto_montaje/

4. Probar agregar archivos 
./mkfs.bwfs fs_data/
mkdir punto_montaje
./mount.bwfs fs_data/ punto_montaje/

Esto es en otra consola
echo "Texto nuevo desde fs_write()" > punto_montaje/hola.txt
cat punto_montaje/hola.txt
touch punto_montaje/vacio.txt

5. Probar unlink 
echo "Hola" > punto_montaje/prueba.txt
ls punto_montaje/
rm punto_montaje/prueba.txt
ls punto_montaje/  # Ya no debe estar

6. Probar carpetas de archivos 
mkdir punto_montaje/midirectorio
ls -l punto_montaje/

7. opendir
ls punto_montaje/
cd punto_montaje/

8. rename
touch punto_montaje/archivo1.txt
mv punto_montaje/archivo1.txt punto_montaje/archivo_renombrado.txt
ls punto_montaje/
